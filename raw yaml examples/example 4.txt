custom_fields:
  room: |
    [[[
      const name = 'Kitchen';

      const motionEntity = 'binary_sensor.hue_motion_sensor_1_motion_3';
      const motion = states[motionEntity];
      const active = motion?.state === 'on';

      let changed = 'unknown';
      if (motion?.last_changed) {
        const mins = Math.round((Date.now() - new Date(motion.last_changed)) / 60000);
        changed = mins < 60 ? `${mins}m ago` : `${Math.round(mins / 60)}h ago`;
      }

      const line1 = active ? 'Motion' : 'No motion';
      const line2 = active ? 'now' : changed;
      const dotColour = active ? '#ff3d8b' : '#30c94a';

      return `
        <div style="min-width:0;overflow:hidden;">
          <div style="font-size:24px;font-weight:700;line-height:1;color:#1f2933;white-space:nowrap;overflow:hidden;text-overflow:ellipsis;">
            ${name}
          </div>
          <div style="display:flex;align-items:flex-start;gap:6px;margin-top:5px;min-width:0;color:#667085;">
            <span style="color:${dotColour};font-size:13px;line-height:1.1;">●</span>
            <div style="font-size:12px;line-height:1.15;min-width:0;overflow:hidden;">
              <div style="white-space:nowrap;overflow:hidden;text-overflow:ellipsis;">${line1}</div>
              <div style="white-space:nowrap;overflow:hidden;text-overflow:ellipsis;">${line2}</div>
            </div>
          </div>
        </div>
      `;
    ]]]
  temp: |
    [[[
      const s = states['sensor.hue_motion_sensor_1_temperature_3'];
      const raw = s ? Number(s.state) : NaN;
      const val = Number.isFinite(raw) ? Math.round(raw) : '–';

      return `
        <div style="text-align:center;min-width:0;overflow:hidden;">
          <ha-icon icon="mdi:thermometer" style="width:22px;height:22px;color:#1d9bf0;"></ha-icon>
          <div style="font-size:18px;font-weight:700;color:#1f2933;line-height:1.05;margin-top:3px;white-space:nowrap;overflow:hidden;text-overflow:clip;">${val}°</div>
          <div style="font-size:11px;color:#667085;line-height:1.1;">°C</div>
        </div>
      `;
    ]]]
  lights: |
    [[[
      const areaId = 'kitchen';

      const lightEntities = Object.keys(states).filter((entityId) => {
        if (!entityId.startsWith('light.')) return false;

        const entityRegistry = hass.entities?.[entityId];
        const deviceId = entityRegistry?.device_id;
        const deviceRegistry = deviceId ? hass.devices?.[deviceId] : undefined;

        const entityArea = entityRegistry?.area_id;
        const deviceArea = deviceRegistry?.area_id;
        const assignedArea = entityArea || deviceArea;

        return assignedArea === areaId;
      });

      const on = lightEntities.filter((entityId) => states[entityId]?.state === 'on').length;
      const total = lightEntities.length;

      return `
        <div style="text-align:center;min-width:0;overflow:hidden;">
          <ha-icon icon="mdi:lightbulb-outline" style="width:22px;height:22px;color:#d4a600;"></ha-icon>
          <div style="font-size:18px;font-weight:700;color:#1f2933;line-height:1.05;margin-top:3px;white-space:nowrap;overflow:hidden;text-overflow:clip;">${on}/${total}</div>
          <div style="font-size:11px;color:#667085;line-height:1.1;">Lights</div>
        </div>
      `;
    ]]]
  humidity: |
    [[[
      const s = states['sensor.dyson_pure_humidity'];
      const raw = s ? Number(s.state) : NaN;
      const val = Number.isFinite(raw) ? Math.round(raw) : '–';

      return `
        <div style="text-align:center;min-width:0;overflow:hidden;">
          <ha-icon icon="mdi:water-percent" style="width:22px;height:22px;color:#1d9bf0;"></ha-icon>
          <div style="font-size:18px;font-weight:700;color:#1f2933;line-height:1.05;margin-top:3px;white-space:nowrap;overflow:hidden;text-overflow:clip;">${val}%</div>
          <div style="font-size:11px;color:#667085;line-height:1.1;">Humidity</div>
        </div>
      `;
    ]]]
  air: |
    [[[
      const s = states['sensor.dyson_pure_pm_2_5'];
      const raw = s ? Number(s.state) : NaN;
      const val = Number.isFinite(raw) ? Math.round(raw) : '–';

      let colour = '#30c94a';
      let icon = 'mdi:air-filter';

      if (Number.isFinite(raw)) {
        if (raw > 35) {
          colour = '#ff5c5c';
          icon = 'mdi:alert-circle-outline';
        } else if (raw > 12) {
          colour = '#d4a600';
        }
      }

      return `
        <div style="text-align:center;min-width:0;overflow:hidden;">
          <ha-icon icon="${icon}" style="width:22px;height:22px;color:${colour};"></ha-icon>
          <div style="font-size:18px;font-weight:700;color:#1f2933;line-height:1.05;margin-top:3px;white-space:nowrap;overflow:hidden;text-overflow:clip;">${val}</div>
          <div style="font-size:11px;color:#667085;line-height:1.1;">PM2.5</div>
        </div>
      `;
    ]]]
