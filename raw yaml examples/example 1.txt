    [[[
      const lightEntities = Object.keys(states).filter((entityId) => {
        if (!entityId.startsWith('light.')) return false;

        const exclude = [
          // Add light groups here if your count looks inflated.
          // 'light.kitchen',
          // 'light.living_room',
          // 'light.bedroom'
        ];

        return !exclude.includes(entityId);
      });

      const on = lightEntities.filter((entityId) => states[entityId]?.state === 'on').length;
      const total = lightEntities.length;

      return `
        <div style="text-align:center;min-width:0;overflow:hidden;">
          <ha-icon icon="mdi:lightbulb-group-outline" style="width:22px;height:22px;color:#ffd84d;"></ha-icon>

        </div>
      `;
    ]]]
  occupancy: |
    [[[
      const people = [
        'person.lewis_pearson',
        'person.megan_joseph'
      ];

      let count = people.filter(p => states[p]?.state === 'home').length;

      const zoneHome = Number(states['zone.home']?.state);
      if (Number.isFinite(zoneHome)) {
        count = zoneHome;
      }

      const occupiedSensor = states['binary_sensor.house_occupied']?.state;
      const occupied = occupiedSensor === 'on' || count > 0;

      const val = Number.isFinite(count) ? count : occupied ? 'Yes' : '0';
      const label = occupied ? 'Home' : 'Away';

      let icon = 'mdi:home-account';
      let colour = occupied ? '#49d44f' : '#8d96a6';

      if (!occupied) {
        icon = 'mdi:home-outline';
      } else if (count >= 2) {
        icon = 'mdi:account-group';
      } else {
        icon = 'mdi:account';
      }

      return `
        <div style="text-align:center;min-width:0;overflow:hidden;">
          <ha-icon icon="${icon}" style="width:22px;height:22px;color:${colour};"></ha-icon>
          <div style="font-size:18px;font-weight:650;color:#f5f7fb;line-height:1.05;margin-top:3px;white-space:nowrap;overflow:hidden;text-overflow:clip;">${val}</div>

        </div>
      `;
    ]]]
